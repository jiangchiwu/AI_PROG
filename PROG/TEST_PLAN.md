# 详细测试计划

## 一、测试策略概述

### 1.1 测试目标

本测试计划旨在全面验证多功能编程器的功能、性能和稳定性，确保其能够支持所有声明的芯片和协议。

**核心目标**：
- 验证所有7种编程协议的功能正确性
- 验证339款芯片的兼容性
- 验证性能指标达到设计要求
- 发现并修复所有已知问题
- 确保系统稳定性和可靠性

### 1.2 测试范围

```

│   ├── GPIO抽象层测试
```

### 1.3 测试方法

**黑盒测试**：
- 仅关注输入和输出
- 不涉及内部实现
- 验证功能正确性

**白盒测试**：
- 了解内部实现细节
- 验证代码覆盖率
- 发现潜在问题

**灰盒测试**：
- 结合黑盒和白盒
- 关注关键路径
- 平衡测试效率和覆盖度

```python
#!/usr/bin/env python3
"""
IIC协议自动化测试脚本
"""
import serial
import time
import struct

class IICTest:
    def __init__(self, port, baudrate=115200):
        self.ser = serial.Serial(port, baudrate, timeout=1)

    def send_command(self, cmd, data=b''):
        """发送命令"""
        frame = struct.pack('<BBH', 0xAA, cmd, len(data)) + data
        frame += bytes([sum(frame) & 0xFF])
        self.ser.write(frame)
        response = self.ser.read(1024)
        return response

    def test_scan(self):
        """设备扫描测试"""
        print("开始IIC设备扫描...")
        result = self.send_command(0x01)  # 扫描命令
        devices = result[4:12]  # 解析设备列表
        print(f"找到设备: {[hex(d) for d in devices if d != 0]}")
        return len([d for d in devices if d != 0]) > 0

    def test_write_read(self, addr, data):
        """写入读取测试"""
        print(f"测试写入读取: 地址={hex(addr)}")
        result = self.send_command(0x02, struct.pack('<IH', addr, len(data)) + data)
        return result[4] == 0x00  # 返回成功标志

    def run_tests(self):
        """运行所有测试"""
        tests = [
            ("设备扫描", self.test_scan),
            ("写入读取", lambda: self.test_write_read(0x0000, b'\x55\xAA')),
        ]

        passed = 0
        for name, test_func in tests:
            try:
                if test_func():
                    print(f"✅ {name} - 通过")
                    passed += 1
                else:
                    print(f"❌ {name} - 失败")
            except Exception as e:
                print(f"❌ {name} - 异常: {e}")

        print(f"\n测试完成: {passed}/{len(tests)} 通过")
        return passed == len(tests)

if __name__ == "__main__":
    tester = IICTest("COM5")
    tester.run_tests()
```

```python
#!/usr/bin/env python3
"""
SPI协议自动化测试脚本
"""
import serial
import time
import struct

class SPITest:
    def __init__(self, port, baudrate=115200):
        self.ser = serial.Serial(port, baudrate, timeout=1)

    def send_command(self, cmd, data=b''):
        """发送命令"""
        frame = struct.pack('<BBH', 0xAA, cmd, len(data)) + data
        frame += bytes([sum(frame) & 0xFF])
        self.ser.write(frame)
        response = self.ser.read(1024)
        return response

    def test_read_id(self):
        """读取JEDEC ID"""
        print("读取Flash ID...")
        result = self.send_command(0x10)  # 读取ID命令
        idcode = result[4:8]
        print(f"Flash ID: {' '.join([hex(b) for b in idcode])}")
        return idcode == b'\xEF\x40\x18'  # W25Q128 ID

    def test_erase_write(self, addr, size):
        """擦除写入测试"""
        print(f"测试擦除写入: 地址={hex(addr)}, 大小={size}")
        test_data = bytes([i % 256 for i in range(size)])

        # 擦除
        erase_cmd = struct.pack('<I', addr)
        result = self.send_command(0x11, erase_cmd)  # 擦除命令
        time.sleep(0.5)  # 等待擦除完成

        # 写入
        write_cmd = struct.pack('<I', addr) + test_data
        result = self.send_command(0x12, write_cmd)  # 写入命令

        # 读取验证
        read_cmd = struct.pack('<II', addr, size)
        result = self.send_command(0x13, read_cmd)  # 读取命令
        read_data = result[8:8+size]

        return read_data == test_data

    def run_tests(self):
        """运行所有测试"""
        tests = [
            ("读取ID", self.test_read_id),
            ("擦除写入", lambda: self.test_erase_write(0x0000, 256)),
        ]

        passed = 0
        for name, test_func in tests:
            try:
                if test_func():
                    print(f"✅ {name} - 通过")
                    passed += 1
                else:
                    print(f"❌ {name} - 失败")
            except Exception as e:
                print(f"❌ {name} - 异常: {e}")

        print(f"\n测试完成: {passed}/{len(tests)} 通过")
        return passed == len(tests)

if __name__ == "__main__":
    tester = SPITest("COM5")
    tester.run_tests()
```
